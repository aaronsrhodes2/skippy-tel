        self.ttl = ttl
        self.clock = clock

    def put(self, key, value):

        # Step 1: remove expired entries
        for k in list(self.cache.keys()):
            stored_at = self.cache[k][1]
            if current_time - stored_at >= self.ttl:
                del self.cache[k]
        # Step 2: check if key is present, and delete if so
        if key in self.cache:
            del self.cache[key]
        # Step 3: evict LRU if needed
        while len(self.cache) >= self.capacity:
            # evict the first key (LRU)
            lru_key = next(iter(self.cache))
            del self.cache[lru_key]
        # Step 4: add the new key
        self.cache[key] = (value, current_time)

    def get(self, key, default=None):
        current_time = self.clock()
        if key in self.cache:
            stored_at = self.cache[key][1]
            if current_time - stored_at >= self.ttl:
                # expired, remove and return default
                del self.cache[key]
                return default
            else:
                # not expired, return value and move to end
                self.cache.move_to_end(key, last=True)
                return self.cache[key][0]
        else:
            return default

    def __len__(self):
        current_time = self.clock()
        count = 0

            stored_at = self.cache[key][1]
            if current_time - stored_at < self.ttl:
                count +=1

            if current_time - stored_at < self.ttl:

                del self.cache[key]
                return value
            else:
                return default
        else:
            return default

        self.ttl = ttl
        self.clock = clock

        current_time = self.clock()

        for k in list(self.cache.keys()):
            stored_at = self.cache[k][1]
            if current_time - stored_at >= self.ttl:
                del self.cache[k]
        # Step 2: check if key is present and delete it
        if key in self.cache:
            del self.cache[key]
        # Step 3: evict LRU if needed
        while len(self.cache) >= self.capacity:
            lru_key = next(iter(self.cache))
            del self.cache[lru_key]
        # Step 4: add the new key
        self.cache[key] = (value, current_time)
