def evaluate(expr: str) -> float:
    """
    Evaluate an arithmetic expression string without using eval, exec, compile, or ast.
    
    Supports: integers, decimals, +, -, *, /, %, **, unary + and -, parentheses.
    Precedence and associativity follow Python's rules.
    """
    
    class Token:
        NUMBER = 'NUMBER'
        PLUS = '+'
        MINUS = '-'
        MULT = '*'
        DIV = '/'
        MOD = '%'
        POW = '**'
        LPAREN = '('
        RPAREN = ')'
        EOF = 'EOF'
        
        def __init__(self, type_, value=None):
            self.type = type_
            self.value = value
        
        def __repr__(self):
            return f'Token({self.type}, {self.value!r})'
    
    class Lexer:
        def __init__(self, text):
            self.text = text
            self.pos = 0
            self.length = len(text)
        
        def skip_spaces(self):
            while self.pos < self.length and self.text[self.pos] == ' ':
                self.pos += 1
        
        def read_number(self):
            start = self.pos
            has_dot = False
            while self.pos < self.length:
                ch = self.text[self.pos]
                if ch.isdigit():
                    self.pos += 1
                elif ch == '.':
                    if has_dot:
                        raise ValueError(f"Invalid number: {self.text[start:self.pos+1]}")
                    has_dot = True
                    self.pos += 1
                else:
                    break
            if self.pos == start:
                raise ValueError("Expected a number")
            return float(self.text[start:self.pos])
        
        def tokenize(self):
            tokens = []
            while self.pos < self.length:
                self.skip_spaces()
                if self.pos >= self.length:
                    break
                ch = self.text[self.pos]
                
                if ch.isdigit() or ch == '.':
                    # Check if it's a dot only (should be part of a number)
                    if ch == '.':
                        # Look ahead to see if there's a digit after
                        if self.pos + 1 < self.length and self.text[self.pos + 1].isdigit():
                            val = self.read_number()
                            tokens.append(Token(Token.NUMBER, val))
                        else:
                            raise ValueError(f"Invalid character: {ch}")
                    else:
                        val = self.read_number()
                        tokens.append(Token(Token.NUMBER, val))
                elif ch == '+':
                    tokens.append(Token(Token.PLUS, '+'))
                    self.pos += 1
                elif ch == '-':
                    tokens.append(Token(Token.MINUS, '-'))
                    self.pos += 1
                elif ch == '*':
                    if self.pos + 1 < self.length and self.text[self.pos + 1] == '*':
                        tokens.append(Token(Token.POW, '**'))
                        self.pos += 2
                    else:
                        tokens.append(Token(Token.MULT, '*'))
                        self.pos += 1
                elif ch == '/':
                    tokens.append(Token(Token.DIV, '/'))
                    self.pos += 1
                elif ch == '%':
                    tokens.append(Token(Token.MOD, '%'))
                    self.pos += 1
                elif ch == '(':
                    tokens.append(Token(Token.LPAREN, '('))
                    self.pos += 1
                elif ch == ')':
                    tokens.append(Token(Token.RPAREN, ')'))
                    self.pos += 1
                else:
                    raise ValueError(f"Unknown character: {ch}")
            
            tokens.append(Token(Token.EOF))
            return tokens
    
    class Parser:
        def __init__(self, tokens):
            self.tokens = tokens
            self.pos = 0
        
        def current(self):
            return self.tokens[self.pos]
        
        def advance(self):
            token = self.tokens[self.pos]
            self.pos += 1
            return token
        
        def expect(self, type_):
            token = self.current()
            if token.type != type_:
                raise ValueError(f"Expected {type_}, got {token.type}")
            return self.advance()
        
        def parse(self):
            result = self.parse_expression()
            if self.current().type != Token.EOF:
                raise ValueError(f"Unexpected token: {self.current()}")
            return result
        
        def parse_expression(self):
            return self.parse_add_sub()
        
        def parse_add_sub(self):
            left = self.parse_mul_div_mod()
            while self.current().type in (Token.PLUS, Token.MINUS):
                op = self.advance()
                right = self.parse_mul_div_mod()
                if op.type == Token.PLUS:
                    left = left + right
                else:
                    left = left - right
            return left
        
        def parse_mul_div_mod(self):
            left = self.parse_power()
            while self.current().type in (Token.MULT, Token.DIV, Token.MOD):
                op = self.advance()
                right = self.parse_power()
                if op.type == Token.MULT:
                    left = left * right
                elif op.type == Token.DIV:
                    if right == 0:
                        raise ZeroDivisionError("division by zero")
                    left = left / right
                else:  # MOD
                    if right == 0:
                        raise ZeroDivisionError("modulo by zero")
                    left = left % right
            return left
        
        def parse_power(self):
            # Right-associative: parse base, then if we see **, parse exponent
            base = self.parse_unary()
            if self.current().type == Token.POW:
                self.advance()
                exponent = self.parse_power()  # Right-associative recursion
                base = base ** exponent
            return base
        
        def parse_unary(self):
            # Handle unary + and -
            if self.current().type == Token.PLUS:
                self.advance()
                return self.parse_unary()
            elif self.current().type == Token.MINUS:
                self.advance()
                return -self.parse_unary()
            else:
                return self.parse_primary()
        
        def parse_primary(self):
            token = self.current()
            if token.type == Token.NUMBER:
                self.advance()
                return token.value
            elif token.type == Token.LPAREN:
                self.advance()
                result = self.parse_expression()
                self.expect(Token.RPAREN)
                return result
            else:
                raise ValueError(f"Unexpected token: {token}")
    
    # Main logic
    if not expr or expr.strip() == '':
        raise ValueError("Empty expression")
    
    lexer = Lexer(expr)
    tokens = lexer.tokenize()
    
    parser = Parser(tokens)
    return parser.parse()
